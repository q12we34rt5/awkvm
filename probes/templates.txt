# probe_id <sigil> <body>
#
# Two sigils, two destinations:
#
#   =>  call-site template. Body is an awk expression with arg0 / arg1 /
#       ... placeholders that codegen substitutes from the IR call's
#       operands at each call site that resolves to the captured
#       mangled name. Result wraps as `dest = expr` or `expr` based on
#       whether the IR call has a dest. Helpers referenced here (e.g.
#       _ostream_int) live in runtime/iostream.awk.
#
#   :=  global-symbol metadata. Body is `<table>=<value>` and codegen
#       at emit_globals_init time populates the named awk table with
#       the value at the global's allocated address. Recognized tables:
#         dest=...   register output target in _STREAM_DEST
#         src=...    register input source in _STREAM_SRC
#
# The probe_id (the suffix on `awkvm_probe_<id>` in probes/src/*.cpp)
# must match for build.rs to pair the captured mangled name with the
# template body.

ostream_int     => _ostream_int(arg0, arg1)
ostream_cstr    => _ostream_cstr(arg0, arg1, arg2)
ostream_double  => _ostream_double(arg0, arg1)

# `long` prints like signed int (printf "%d" works as-is).
ostream_long    => _ostream_int(arg0, arg1)

# `bool` would too, except awkvm's signed integer model represents `i1
# true` as -1 (high bit set, sign-extended). Normalize to 0 / 1 with
# `!= 0` before printing.
ostream_bool    => _ostream_int(arg0, arg1 != 0)

# unsigned needs zext from signed model. The "32" / "64" literals are
# the source-type bit width and bake into the awk template directly
# (no arg substitution); the helper does `val + 2^bits` if val is
# negative in the signed view.
ostream_uint    => _ostream_unsigned(arg0, arg1, 32)
ostream_ulong   => _ostream_unsigned(arg0, arg1, 64)

# void* is zext-then-hex with the libc++ "0x" prefix.
ostream_voidptr => _ostream_voidptr(arg0, arg1)

# os.write(buf, n) — block write of n bytes, no formatting. Body is
# identical to ostream_cstr (libc++'s `<< "literal"` path lowers to
# the same byte-loop primitive); reuse the same helper.
ostream_write   => _ostream_cstr(arg0, arg1, arg2)

# os.put(c) — write one byte, return the ostream for chaining.
ostream_put     => _ostream_put(arg0, arg1)

# ostream / istream globals. cout stays implicit (anything not in
# _STREAM_DEST defaults to stdout). cin currently is the only istream
# we recognize; sstream / fstream additions go through the same table.
ostream_cerr    := dest=/dev/stderr
ostream_clog    := dest=/dev/stderr
istream_cin     := src=/dev/stdin

# cin >> int / long. arg1 is the IR destination pointer; the helper
# reads one whitespace-delimited token from the istream's line buffer
# and stores into it via `_store(addr, val, bits)`.
istream_int     => _istream_int(arg0, arg1, 32)
istream_long    => _istream_int(arg0, arg1, 64)

# cin >> unsigned / unsigned long. Token-read, awk parses leading
# digits non-negatively, then _istream_unsigned wraps high values
# into the awkvm signed integer model before `_store(addr, v, bits)`.
istream_uint    => _istream_unsigned(arg0, arg1, 32)
istream_ulong   => _istream_unsigned(arg0, arg1, 64)

# cin >> double. Same token-read flow, but the destination is an
# 8-byte IEEE 754 slot, so the helper goes through `_store_f64`.
istream_double  => _istream_double(arg0, arg1)

# is.read(buf, n) — block read of up to n bytes. Stores the actual
# count at MEM[stream+8] so the call-site-inlined `is.gcount()`
# (clang -O1 collapses it to a straight load from that offset)
# returns the right value.
istream_read    => _istream_read(arg0, arg1, arg2)

# is.get() — read one byte, return -1 on EOF. Same gcount-via-MEM
# convention as read.
istream_get     => _istream_get(arg0)

# std::ofstream / std::ifstream constructor — register the file as a
# write / read stream against the ofstream/ifstream's `this` pointer
# (arg0). Subsequent `<<` / `>>` operations on the object resolve
# through the same _ostream_* / _istream_* helpers as cout / cin
# because they're inherited from the base class. Destructors aren't
# probed (virtual dispatch makes the mangled name awkward to capture
# from a probe body) — gawk's process-exit auto-flush closes the
# file in practice.
ofstream_ctor   => _ofstream_open(arg0, _cstr(arg1))
ifstream_ctor   => _ifstream_open(arg0, _cstr(arg1))

# basic_filebuf::close — explicit user `f.close()` and the close
# inside the destructor's linkonce_odr body both route here. arg0 is
# the filebuf's `this` (= stream addr + 8 in libc++), which matches
# the +8 entry registered by `_o/ifstream_open`.
filebuf_close   => _stream_close(arg0)
